# 协方差交互实验

> 拖拽散点，直观感受**协方差的正 / 负 / 零、绝对值大小，以及"单位影响 → 相关系数"**。全部在浏览器本地运行，零依赖。

## 指令

- **拖动任意圆点**，散点图会发生-> 右侧实时更新协方差 $\Sigma_{12}$ 与相关系数 $ρ$。
- 想快速看某种形态：点下面的 **正相关 / 负相关 / 无相关** 预设。
- 点 **固定单位** 里的 `cm ↔ m`，观察：协方差的**数值**变了，但相关系数 $ρ$ **不变**——这就是"归一化后跨单位可比"的原因。

<script>
(function () {
  var svg = document.getElementById('covlab-svg');
  var PAD = { l: 46, t: 14, r: 14, b: 30 };
  var W = 560, H = 360;
  var pW = W - PAD.l - PAD.r, pH = H - PAD.t - PAD.b;
  var x0 = 150, x1 = 190, y0 = 40, y1 = 100; // cm / kg 取值范围
  function X(v) { return PAD.l + (v - x0) / (x1 - x0) * pW; } // cm->px
  function Y(v) { return PAD.t + (1 - (v - y0) / (y1 - y0)) * pH; } // kg->px(向下)
  function XI(px) { return (px - PAD.l) / pW * (x1 - x0) + x0; }
  function YI(py) { return (1 - (py - PAD.t) / pH) * (y1 - y0) + y0; }

  var NS = 'http://www.w3.org/2000/svg';
  var rng = Math.random;
  var points = [];

  function gen(mode) {
    var n = 12, a = [];
    if (mode === 'pos') {
      for (var i = 0; i < n; i++) {
        var h = 152 + rng() * 36;
        var w = 42 + (h - 152) * 1.25 + (rng() - 0.5) * 7;
        a.push({ h: h, w: Math.max(40, Math.min(100, w)) });
      }
    } else if (mode === 'neg') {
      for (var i2 = 0; i2 < n; i2++) {
        var h2 = 152 + rng() * 36;
        var w2 = 96 - (h2 - 152) * 1.25 + (rng() - 0.5) * 7;
        a.push({ h: h2, w: Math.max(40, Math.min(100, w2)) });
      }
    } else if (mode === 'none') {
      for (var i3 = 0; i3 < n; i3++) { a.push({ h: 170 + (rng() - 0.5) * 20, w: 70 + (rng() - 0.5) * 30 }); }
    } else { // 乱点：真正的均匀独立
      for (var i4 = 0; i4 < n; i4++) { a.push({ h: x0 + rng() * (x1 - x0), w: y0 + rng() * (y1 - y0) }); }
    }
    return a;
  }

  function drawAxes() {
    var g = document.createElementNS(NS, 'g');
    for (var xg = 155; xg < 195; xg += 5) {
      addLine(g, X(xg), PAD.t, X(xg), PAD.t + pH, '#eef1f6');
      addText(g, (X(xg)), PAD.t + pH + 15, xg, '#8a94a6', 10, 'middle');
    }
    for (var yg = 50; yg < 95; yg += 10) {
      addLine(g, PAD.l, Y(yg), PAD.l + pW, Y(yg), '#eef1f6');
      addText(g, PAD.l - 8, Y(yg) + 3, yg, '#8a94a6', 10, 'end');
    }
    addRect(g, PAD.l, PAD.t, pW, pH);
    addText(g, PAD.l + pW / 2, H - 6, '体重 / kg', '#44506a', 12, 'middle');
    var l = document.createElementNS(NS, 'text');
    l.setAttribute('x', 14); l.setAttribute('y', PAD.t + pH / 2);
    l.setAttribute('font-size', 12); l.setAttribute('fill', '#44506a');
    l.setAttribute('text-anchor', 'middle');
    l.setAttribute('transform', 'rotate(-90 14 ' + (PAD.t + pH / 2) + ')');
    l.textContent = '身高 / cm';
    g.appendChild(l);
    svg.appendChild(g);
  }

  function el(tag, attrs, text) {
    var e = document.createElementNS(NS, tag);
    for (var k in attrs) e.setAttribute(k, attrs[k]);
    if (text != null) e.textContent = text;
    return e;
  }
  function addLine(g, x1, y1, x2, y2, stroke) { var e = el('line', { x1: x1, y1: y1, x2: x2, y2: y2, stroke: stroke }); g.appendChild(e); }
  function addText(g, x, y, t, fill, fs, anchor) { g.appendChild(el('text', { x: x, y: y, 'font-size': fs, fill: fill, 'text-anchor': anchor }, t)); }
  function addRect(g, x, y, w, h) { g.appendChild(el('rect', { x: x, y: y, width: w, height: h, fill: 'none', stroke: '#d5dbe6' })); }

  function drawPoints() {
    var old = svg.querySelector('#covlab-fit'); if (old) old.remove();
    var oldp = svg.querySelector('#covlab-pts'); if (oldp) oldp.remove();
    var s = calc();
    // 拟合线
    var fit = el('g', { id: 'covlab-fit' });
    var slope = s.cov / s.vh;
    function ff(h) { return s.mw + slope * (h - s.mh); }
    addLine(fit, X(x0), Y(ff(x0)), X(x1), Y(ff(x1)), 'rgba(76,123,240,.55)');
    svg.appendChild(fit);
    // 散点
    var g = el('g', { id: 'covlab-pts' });
    points.forEach(function (p, i) {
      g.appendChild(el('circle', {
        cx: X(p.h), cy: Y(p.w), r: 6, id: 'covlab-dot-' + i,
        fill: p.h === drag && p.w === dragw ? '#d6336c' : '#2f6fed',
        'fill-opacity': 0.9, stroke: '#fff', 'stroke-width': 1.5,
        style: 'cursor:grab'
      }));
    });
    svg.appendChild(g);
  }

  function calc() {
    var n = points.length;
    if (!n) return { mh: 0, mw: 0, vh: 0, vw: 0, cov: 0, r: 0 };
    var mh = 0, mw = 0;
    points.forEach(function (p) { mh += p.h; mw += p.w; });
    mh /= n; mw /= n;
    var vh = 0, vw = 0, cov = 0;
    points.forEach(function (p) {
      vh += (p.h - mh) * (p.h - mh);
      vw += (p.w - mw) * (p.w - mw);
      cov += (p.h - mh) * (p.w - mw);
    });
    vh /= (n - 1); vw /= (n - 1); cov /= (n - 1);
    var s = Math.sqrt(vh * vw);
    var r = s > 0 ? cov / s : 0;
    return { mh: mh, mw: mw, vh: vh, vw: vw, cov: cov, r: r };
  }

  var unitM = false;
  function fmt(x, d) { return (x == null) ? '—' : Number(x).toFixed(d); }
  function renderReadout() {
    var s = calc();
    if (unitM) {
      document.getElementById('covlab-cov').textContent = fmt(s.cov * 1e-4, 4) + ' m·kg';
      document.getElementById('covlab-vh').textContent = fmt(s.vh * 1e-4, 4) + ' m²';
    } else {
      document.getElementById('covlab-cov').textContent = fmt(s.cov, 2) + ' cm·kg';
      document.getElementById('covlab-vh').textContent = fmt(s.vh, 2) + ' cm²';
    }
    document.getElementById('covlab-vw').textContent = fmt(s.vw, 2) + ' kg²';
    var r = s.r;
    document.getElementById('covlab-r').textContent = fmt(r, 3);
    var tag = '';
    if (r > 0.3) tag = '正相关：同增同减';
    else if (r < -0.3) tag = '负相关：此消彼长';
    else tag = '几乎无线性关系';
    document.getElementById('covlab-tag').textContent = tag;
  }

  var drag = null, dragw = null;
  function onDown(ev) {
    var rect = svg.getBoundingClientRect();
    var sx = (ev.clientX - rect.left) / rect.width * W;
    var sy = (ev.clientY - rect.top) / rect.height * H;
    var best = null, bd = 18;
    points.forEach(function (p, i) {
      var dx = X(p.h) - sx, dy = Y(p.w) - sy, d = Math.hypot(dx, dy);
      if (d < bd) { bd = d; best = p; }
    });
    if (best) { drag = best; dragw = best; return; }
    // 没点到点：在最近空白处新增一个点
    var h = Math.max(x0, Math.min(x1, XI(sx)));
    var w = Math.max(y0, Math.min(y1, YI(sy)));
    points.push({ h: h, w: w });
    drag = points[points.length - 1]; dragw = drag;
  }
  function onMove(ev) {
    if (!drag) return;
    ev.preventDefault();
    var rect = svg.getBoundingClientRect();
    var sx = (ev.clientX - rect.left) / rect.width * W;
    var sy = (ev.clientY - rect.top) / rect.height * H;
    drag.h = Math.max(x0, Math.min(x1, XI(sx)));
    drag.w = Math.max(y0, Math.min(y1, YI(sy)));
    drawPoints(); renderReadout();
  }
  function onUp() { drag = null; dragw = null; drawPoints(); }

  svg.addEventListener('pointerdown', onDown);
  svg.addEventListener('pointermove', onMove);
  window.addEventListener('pointerup', onUp);

  document.querySelectorAll('#covlab [data-mode]').forEach(function (b) {
    b.addEventListener('click', function () {
      points = gen(b.getAttribute('data-mode'));
      drawPoints(); renderReadout();
    });
  });
  var cb = document.getElementById('covlab-unit');
  cb.addEventListener('change', function () {
    unitM = cb.checked;
    renderReadout();
    document.getElementById('covlab-unitnote').style.display = unitM ? 'inline' : 'none';
  });

  function init() {
    svg.innerHTML = '';
    drawAxes();
    points = gen('pos');
    drawPoints(); renderReadout();
  }
  init();
})();
</script>

## 你看到了什么

- 右上↗方向拉伸 = **$ρ$ 接近 +1**，标签显示"正相关：同增同减"。
- 左上↘方向 = **$ρ$ 接近 -1**，标签显示"负相关：此消彼长"。
- 点填满成一团 = **$|ρ|$ 接近 0**，标签显示"几乎无线性关系"。
- 切 **身高用 m**：协方差数值直接小了四个数量级（$0.01^2$），但相关系数 $ρ$ **一个数字都不变**——这正是它被发明出来、能被跨单位/跨变量比较的原因。
